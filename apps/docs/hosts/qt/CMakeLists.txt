cmake_minimum_required(VERSION 3.24)
project(ForaOfficeDocsQt VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets WebEngineWidgets WebChannel Network PrintSupport)
add_library(ForaOfficeDocsQt STATIC DocsQtEditor.cpp DocsQtEditor.hpp DocsQtPrinting.cpp DocsQtPrinting.hpp)
target_include_directories(ForaOfficeDocsQt PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ForaOfficeDocsQt PUBLIC Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel Qt6::Network Qt6::PrintSupport)
if(MSVC)
  target_compile_options(ForaOfficeDocsQt PRIVATE /W4 /utf-8 /permissive-)
endif()
add_executable(docs-qt-host main.cpp)
target_link_libraries(docs-qt-host PRIVATE ForaOfficeDocsQt)
if(MSVC)
  target_compile_options(docs-qt-host PRIVATE /W4 /utf-8 /permissive-)
endif()
