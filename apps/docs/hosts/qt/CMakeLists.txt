cmake_minimum_required(VERSION 3.24)
project(ForaOfficeDocsQt VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets WebEngineWidgets WebChannel Network PrintSupport)
add_library(ForaOfficeDocsQt STATIC DocsQtEditor.cpp DocsQtEditor.hpp DocsQtBackend.cpp DocsQtBackend.hpp DocsQtPrinting.cpp DocsQtPrinting.hpp)
target_include_directories(ForaOfficeDocsQt PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ForaOfficeDocsQt PUBLIC Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel Qt6::Network Qt6::PrintSupport)
if(MSVC)
  target_compile_options(ForaOfficeDocsQt PRIVATE /W4 /utf-8 /permissive-)
endif()
add_executable(docs-qt-host main.cpp)
target_link_libraries(docs-qt-host PRIVATE ForaOfficeDocsQt)
if(MSVC)
  target_compile_options(docs-qt-host PRIVATE /W4 /utf-8 /permissive-)
endif()

option(FORAOFFICE_DOCS_QT_BUILD_TESTS "Build the SDK's native Qt regression tests" OFF)
if(FORAOFFICE_DOCS_QT_BUILD_TESTS)
  enable_testing()
  find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
  find_program(DOCS_TEST_NODE NAMES node REQUIRED)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../services/node-host.cjs")
    set(test_package_default "${CMAKE_CURRENT_SOURCE_DIR}/../..")
  else()
    set(test_package_default "${CMAKE_CURRENT_SOURCE_DIR}/../../out/sdk")
  endif()
  set(DOCS_TEST_PACKAGE_ROOT "${test_package_default}" CACHE PATH "Built Docs SDK used by the native tests")
  add_executable(docs-qt-startup-test tests/startup.cpp)
  target_link_libraries(docs-qt-startup-test PRIVATE ForaOfficeDocsQt)
  if(MSVC)
    target_compile_options(docs-qt-startup-test PRIVATE /W4 /utf-8 /permissive-)
  endif()
  add_test(NAME docs-qt-gui-independent-startup COMMAND docs-qt-startup-test
    --package "${DOCS_TEST_PACKAGE_ROOT}" --node "${DOCS_TEST_NODE}")
  set_tests_properties(docs-qt-gui-independent-startup PROPERTIES TIMEOUT 30)
  add_test(NAME docs-qt-nonblocking-disposal COMMAND docs-qt-startup-test
    --package "${DOCS_TEST_PACKAGE_ROOT}" --node "${DOCS_TEST_NODE}" --hold-exit)
  set_tests_properties(docs-qt-nonblocking-disposal PROPERTIES TIMEOUT 30)
  add_executable(docs-qt-backend-test tests/backend.cpp)
  target_link_libraries(docs-qt-backend-test PRIVATE ForaOfficeDocsQt Qt6::Test)
  if(MSVC)
    target_compile_options(docs-qt-backend-test PRIVATE /W4 /utf-8 /permissive-)
  endif()
  add_test(NAME docs-qt-backend-lifecycle COMMAND docs-qt-backend-test -o -,txt)
  set_tests_properties(docs-qt-backend-lifecycle PROPERTIES TIMEOUT 60
    ENVIRONMENT "FORAOFFICE_QT_TEST_NODE=${DOCS_TEST_NODE}"
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Test>")
  add_executable(docs-qt-services-test tests/services.cpp)
  target_link_libraries(docs-qt-services-test PRIVATE ForaOfficeDocsQt Qt6::Test)
  if(MSVC)
    target_compile_options(docs-qt-services-test PRIVATE /W4 /utf-8 /permissive-)
  endif()
  add_test(NAME docs-qt-service-overrides COMMAND docs-qt-services-test -o -,txt)
  set_tests_properties(docs-qt-service-overrides PROPERTIES TIMEOUT 30
    ENVIRONMENT "FORAOFFICE_QT_TEST_NODE=${DOCS_TEST_NODE};FORAOFFICE_QT_TEST_PACKAGE=${DOCS_TEST_PACKAGE_ROOT}"
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Test>")
  add_executable(docs-qt-sdk-test tests/sdk.cpp)
  target_link_libraries(docs-qt-sdk-test PRIVATE ForaOfficeDocsQt Qt6::Test)
  if(MSVC)
    target_compile_options(docs-qt-sdk-test PRIVATE /W4 /utf-8 /permissive-)
  endif()
  add_test(NAME docs-qt-public-sdk COMMAND docs-qt-sdk-test -o "${CMAKE_BINARY_DIR}/sdk-test-results.txt,txt")
  set_tests_properties(docs-qt-public-sdk PROPERTIES TIMEOUT 180
    ENVIRONMENT "FORAOFFICE_QT_TEST_NODE=${DOCS_TEST_NODE};FORAOFFICE_QT_TEST_PACKAGE=${DOCS_TEST_PACKAGE_ROOT};FORAOFFICE_QT_TEST_RESULTS=${CMAKE_BINARY_DIR}/sdk-test-output"
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Test>")
  add_executable(docs-qt-mcp-test tests/mcp.cpp)
  target_link_libraries(docs-qt-mcp-test PRIVATE ForaOfficeDocsQt Qt6::Test)
  if(MSVC)
    target_compile_options(docs-qt-mcp-test PRIVATE /W4 /utf-8 /permissive-)
  endif()
  add_test(NAME docs-qt-http-mcp COMMAND docs-qt-mcp-test -o "${CMAKE_BINARY_DIR}/mcp-test-results.txt,txt")
  set_tests_properties(docs-qt-http-mcp PROPERTIES TIMEOUT 180
    ENVIRONMENT "FORAOFFICE_QT_TEST_NODE=${DOCS_TEST_NODE};FORAOFFICE_QT_TEST_PACKAGE=${DOCS_TEST_PACKAGE_ROOT}"
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Test>")
endif()
